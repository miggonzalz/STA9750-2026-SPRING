library(purrr)

slow_download <- purrr::slowly(
  download.file,
  purrr::rate_delay(pause = 5)
)

read_olympedia <- function(url, ...){
  if(stringr::str_starts(url, "/")) {
    url <- stringr::str_remove(url, "^/")
  }

  cache_dir <- fs::path("data", "mp04", "olympedia_cache")
  if(!fs::dir_exists(cache_dir)) fs::dir_create(cache_dir, recurse = TRUE)

  cache_path <- fs::path(
    cache_dir,
    url |> as.character() |> stringr::str_replace_all("/", "_")
  )

  if(!fs::file_exists(cache_path)){
    src_url <- paste0("https://www.olympedia.org/", url)
    slow_download(src_url, cache_path, mode = "wb")
  }

  rvest::read_html(cache_path)
}

roman_to_integer <- function(x) {
  suppressWarnings(as.integer(as.roman(x)))
}

# This chunk scrapes the main Olympedia editions page and builds a tidy table of Olympic editions with year, city, season, and edition ID.
read_editions <- function() {
  pg <- read_olympedia("editions")

  tables <- pg |> html_elements("table")

  parse_editions_table <- function(tbl, season_label) {
    rows <- tbl |> html_elements("tr")

    map_dfr(rows, function(row) {
      tds <- row |> html_elements("td")
      if(length(tds) < 8) return(NULL)

      edition_link <- row |> html_element("a[href^='/editions/']")
      edition_url <- if(length(edition_link) == 0) NA_character_ else html_attr(edition_link, "href")
      edition_id  <- str_extract(edition_url, "\\d+$") |> as.integer()

      olympiad_roman <- tds[[1]] |> html_text2() |> str_trim()
      year_txt <- tds[[2]] |> html_text2() |> str_trim()
      city <- tds[[3]] |> html_text2() |> str_trim()
      note <- tds[[8]] |> html_text2() |> str_squish()

      year <- suppressWarnings(as.integer(year_txt))

      flag_img <- tds[[4]] |> html_element("img")
      flag_src <- if(length(flag_img) == 0) NA_character_ else html_attr(flag_img, "src")
      host_country_code <- str_extract(flag_src, "(?<=/)[A-Z]{3}(?=\\.png$)")

      if(is.na(edition_id) || is.na(year) || olympiad_roman == "") return(NULL)

      tibble(
        season = season_label,
        year = year,
        city = city,
        olympiad_roman = olympiad_roman,
        olympiad_id = roman_to_integer(olympiad_roman),
        edition_id = edition_id,
        edition_url = edition_url,
        host_country_code = host_country_code,
        note = note
      )
    })
  }

  bind_rows(
    parse_editions_table(tables[[1]], "Summer"),
    parse_editions_table(tables[[2]], "Winter")
  ) |>
    distinct(edition_id, .keep_all = TRUE) |>
    filter(!str_detect(str_to_lower(note), "not held")) |>
    select(-note) |>
    arrange(year, season)
}

#This chunk joins the edition-level data to the list of disciplines available for each edition.


read_edition_disciplines <- function(edition_url) {
  pg <- read_olympedia(edition_url)

  sport_table <- pg |>
    html_element(xpath = "//h2[normalize-space()='Medal Disciplines']/following-sibling::table[1]")

  if(length(sport_table) == 0) {
    return(tibble(
      discipline = character(),
      discipline_url = character(),
      discipline_code = character()
    ))
  }

  links <- sport_table |> html_elements("a")

  tibble(
    discipline = html_text2(links) |> str_squish(),
    discipline_url = html_attr(links, "href")
  ) |>
    filter(str_detect(discipline_url, "/sports/")) |>
    mutate(
      discipline_code = str_extract(discipline_url, "(?<=/sports/)[A-Z0-9]+")
    ) |>
    distinct()
}

# This chunk combines all of the discipline medal tables into one master Olympedia medal dataset.
read_discipline_medal_table <- function(discipline_url) {
  pg <- read_olympedia(discipline_url)

  medal_table_node <- pg |>
    html_element(xpath = "//h2[normalize-space()='Medal table']/following-sibling::table[1]")

  if(length(medal_table_node) == 0) {
    return(tibble(
      competitor = character(),
      country_code = character(),
      gold = integer(),
      silver = integer(),
      bronze = integer(),
      total = integer()
    ))
  }

  medal_tbl <- medal_table_node |> html_table(fill = TRUE)

  # At least 6 columns: competitor, 3‑letter code, then Gold, Silver, Bronze, Total
  if(ncol(medal_tbl) < 6) {
    return(tibble(
      competitor = character(),
      country_code = character(),
      gold = integer(),
      silver = integer(),
      bronze = integer(),
      total = integer()
    ))
  }

  tibble(
    competitor = as.character(medal_tbl[[1]]),
    country_code = as.character(medal_tbl[[2]]),
    gold = suppressWarnings(as.integer(medal_tbl[[3]])),
    silver = suppressWarnings(as.integer(medal_tbl[[4]])),
    bronze = suppressWarnings(as.integer(medal_tbl[[5]])),
    total = suppressWarnings(as.integer(medal_tbl[[6]]))
  ) |>
    filter(!is.na(country_code), country_code != "")
}

I used the provided read_olympedia caching setup plus the functions read_editions, read_edition_disciplines, and read_discipline_medal_table to scrape all 59 modern Olympic editions and their medal disciplines and medal tables. I then joined these into a single tidy table olympedia_medals with 8,450 rows, where each row corresponds to one country’s medal performance in a specific sport at a specific Olympic edition. Columns include season, year, edition_id, country_code, discipline, gold, silver, bronze, and total.

  group_by(year, season, country_code) |>
  summarise(
    gold = sum(gold, na.rm = TRUE),
    silver = sum(silver, na.rm = TRUE),
    bronze = sum(bronze, na.rm = TRUE),
    total = sum(total, na.rm = TRUE),
    .groups = "drop"
  ) |>
  inner_join(host_editions, by = c("year", "season")) |>
  filter(country_code == host_country_code)

host_lagged <- host_totals |>
  arrange(season, year) |>
  group_by(season) |>
  mutate(
    prev_gold = lag(gold),
    prev_silver = lag(silver),
    prev_bronze = lag(bronze),
    prev_total = lag(total)
  ) |>
  ungroup()

host_change <- host_lagged |>
  mutate(
    pct_gold = if_else(!is.na(prev_gold) & prev_gold > 0, 100 * (gold - prev_gold) / prev_gold, NA_real_),
    pct_silver = if_else(!is.na(prev_silver) & prev_silver > 0, 100 * (silver - prev_silver) / prev_silver, NA_real_),
    pct_bronze = if_else(!is.na(prev_bronze) & prev_bronze > 0, 100 * (bronze - prev_bronze) / prev_bronze, NA_real_),
    pct_total = if_else(!is.na(prev_total) & prev_total > 0, 100 * (total - prev_total) / prev_total, NA_real_)
  ) |>
  select(year, season, country_code, pct_gold, pct_silver, pct_bronze, pct_total)

Final Insights and Deliverable

Team USA is poised for a strong performance at the Los Angeles 2028 Summer Olympics, benefiting from a high-performing historical baseline, home-field advantage, and new medal opportunities in sports like flag football and lacrosse. Using historical Olympic data and Monte Carlo simulations, we predict that Team USA will secure 55–95 Gold, 49–98 Silver, and 45–88 Bronze, for a grand total ranging from 137 to 299 medals. Even if they score on the lower end of our projections, that would still be 21% higher than the last Summer Olympics held in Tokyo and 36% higher than the last time the U.S.A hosted the Summer Olympics back in 1996 Atlanta, Georgia.

This forecast is driven by three key factors:

TipHistorial Domiance: Established Baseline

Consistent success in recent Summer Games highlights the depth of the U.S. athlete pipeline and robust institutional funding.

TipHome Turf: The Host-Country Effect

Historical data indicates that hosting the Games provides a boost in performance, attributed to increased national attention and home support.

TipNew Frontiers & Making History: New Sport Opportunities

The introduction of sports such as baseball/softball, cricket, and squash expands the competitive landscape, creating additional pathways for medal success.
